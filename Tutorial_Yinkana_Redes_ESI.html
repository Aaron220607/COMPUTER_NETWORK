Variables y anotaciones de tipo

Python es dinámico: no declaras el tipo. Pero la yinkana usa anotaciones de tipo (type hints). Son opcionales pero ayudan a entender qué espera cada función.

# Sin anotaciones (Python clasico)
nombre = "fast_rhino"
puerto = 2000

# Con anotaciones de tipo (como en la yinkana)
nombre: str = "fast_rhino"
puerto: int = 2000
datos: bytes = b"hola"   # bytes: cadena de bytes, no texto
activo: bool = True

# Las anotaciones NO hacen nada en ejecucion, son solo documentacion
bytes vs str — LA diferencia más importante

En la yinkana casi todo son bytes, no str. La red trabaja con bytes. Entender la diferencia es clave para no tener errores raros.

# str: texto Unicode para el humano
texto: str = "hola mundo"

# bytes: secuencia de bytes raw, lo que viaja por la red
datos: bytes = b"hola mundo"   # la b delante indica bytes

# Convertir entre ellos
texto_a_bytes = texto.encode()             # str -> bytes (UTF-8)
bytes_a_texto = datos.decode()             # bytes -> str

# En la yinkana lo ves constantemente:
bytes(str(puerto_libre), encoding="utf-8")  # int -> str -> bytes
recibido.decode()                            # para imprimir lo recibido

# Concatenar bytes (no se puede mezclar con str)
mensaje = b"puerto: " + bytes(str(8080), encoding="utf-8")

# Comprobar si empieza por algo
if recibido.startswith(b"identifier:"):
    print("es el identificador!")

# upper() tambien funciona en bytes
b"fast_rhino".upper()   # -> b"FAST_RHINO"

with — context manager

El with garantiza que el socket se cierra aunque haya un error. Es el patrón que usa la yinkana en todos los hitos.

# Sin with (malo): si falla recv, el socket queda abierto
s = socket.socket()
s.connect(("rick", 2000))
datos = s.recv(1024)
s.close()   # si la linea anterior falla, esto no se ejecuta

# Con with (bien): se cierra siempre
with socket.socket() as cliente:
    cliente.connect(("rick", 2000))
    datos = cliente.recv(1024)
# socket cerrado automaticamente aqui

struct — empaquetar datos binarios

Convierte tipos Python a bytes en un formato binario exacto. Esencial para el Hito 5 (protocolo YAP).

import struct

# Formato "!3sHBHH":
# !  = big-endian (formato de red)
# 3s = 3 bytes (string de longitud fija)
# H  = unsigned short = 2 bytes sin signo
# B  = unsigned char  = 1 byte sin signo
# Total cabecera: 3+2+1+2+2 = 10 bytes

# Empaquetar: Python -> bytes
cabecera = struct.pack("!3sHBHH", b"YAP", 0, 0, 0, 1)

# Desempaquetar: bytes -> Python
num_mag, tipo, codigo, suma, seq = struct.unpack("!3sHBHH", mensaje[:10])

# Acceder al resto del mensaje (la carga)
carga = mensaje[10:]   # todo lo que hay despues de los 10 bytes de cabecera

Sockets en Python
La biblioteca socket de Python es la interfaz directa a las llamadas de red del SO. Todo lo que usa la yinkana explicado aquí.
TCP — cliente básico
import socket

# socket.socket() sin argumentos = TCP + IPv4
with socket.socket() as cliente:
    cliente.connect(("rick", 2000))   # (ip, puerto) como tupla

    # sendall garantiza que se envian TODOS los bytes
    cliente.sendall(b"fast_rhino")

    # recv(n) devuelve HASTA n bytes. Puede ser menos.
    recibido = cliente.recv(1024)

# Para recibir un mensaje completo a veces hay que hacer un bucle:
while enunciado is None:
    recibido = cliente.recv(1024)
    if recibido.startswith(b"identifier:"):
        enunciado = recibido
Cuidado con recv(): TCP es un flujo de bytes, no de mensajes. Un recv(1024) puede devolver 10 bytes aunque el servidor haya enviado 500. Por eso en muchos hitos hay un bucle leyendo hasta cumplir una condición.
UDP — sendto y recvfrom
# UDP: hay que pasar type=socket.SOCK_DGRAM
with socket.socket(type=socket.SOCK_DGRAM) as cliente:

    # Enviar a (ip, puerto) - no hay connect() en UDP
    cliente.sendto(mensaje, ("rick", 4000))

    # Recibir: devuelve (datos, (ip_origen, puerto_origen))
    recibido, direccion_origen = cliente.recvfrom(1024)

    # Responder al que nos envio el mensaje
    cliente.sendto(respuesta, direccion_origen)

# UDP + bind: reservar un puerto libre
with socket.socket(type=socket.SOCK_DGRAM) as s:
    s.bind(("", 0))                      # "" = cualquier interfaz, 0 = puerto libre
    puerto_libre = s.getsockname()[1]   # [1] = el puerto asignado
TCP — servidor (bind + listen + accept)
with socket.socket() as servidor:
    servidor.bind(("", 0))                        # puerto libre
    puerto_libre = servidor.getsockname()[1]
    servidor.listen(4)                             # cola de hasta 4 conexiones

    while True:
        peticion, direccion = servidor.accept()   # bloquea hasta nueva conexion
        datos = peticion.recv(1024)
        peticion.sendall(b"respuesta")
        peticion.close()

Por qué bind? En UDP no hay conexión. El servidor necesita saber a qué puerto enviar la respuesta. Al hacer bind() y comunicar ese puerto, el servidor puede responder a nuestra dirección explícitamente.

El servidor envía palabras por TCP. Hay que calcular las longitudes y devolverlas como stream hasta superar 1000.

  • 3

  • 5

def longitudes(tcp_socket: socket.socket, maximo: int) -> bytes:
    suma: int = 0
    stream_numeros: bytes = b" "
    palabra: bytes = b""   # guardamos la palabra incompleta

    while not suma >= maximo:
        recibido = tcp_socket.recv(1024)

        # Juntamos con la palabra anterior (puede estar cortada en el recv)
        lista_palabras = (palabra + recibido).split(b" ")

        while not suma >= maximo and len(lista_palabras) > 1:
            palabra = lista_palabras.pop(0)        # quitar la primera palabra
            suma += len(palabra)
            stream_numeros += bytes(str(len(palabra)), encoding="utf-8") + b" "

        # La ultima puede estar a medias, guardar para el proximo recv
        palabra = lista_palabras.pop(0)

    return stream_numeros

# Se envia todo junto antes de esperar respuesta:
cliente.sendall(identificador + b" " + longitudes(cliente, 1000) + b"--")
El problema del fragmentado TCP: TCP puede partir una palabra por la mitad entre dos recv(). Por eso se guarda la última palabra incompleta y se une con el próximo recv() antes de hacer split.
Hito 3
Cifrado César
El servidor envía palabras cifradas. Hay que encontrar el número de desplazamiento en el stream y descifrar las últimas N palabras.
Hito 3 Descifrado César TCP — rick:6501
  • 1
    Conecta y envía el identificador
  • 2
    Recibe el stream de palabras cifradas hasta encontrar un número X
  • 3
    Retrocede X palabras antes del número en el stream
  • 4
    Descifra esas X palabras desplazando cada letra X posiciones (César circular)
  • 5
    Envía las palabras descifradas + b"--"
def descifrar_chr(c: int, numero: int) -> int:
    # c es un int (codigo ASCII) porque al iterar bytes Python da ints
    # Cifrado Cesar: mover c tantas posiciones a la derecha
    if chr(c).islower():
        if c + numero > ord("z"):       # si se pasa de 'z', vuelve desde 'a'
            return ord("a") + numero - 1 - (ord("z") - c)
        return c + numero
    if chr(c).isupper():
        if c + numero > ord("Z"):
            return ord("A") + numero - 1 - (ord("Z") - c)
        return c + numero
    return c                            # no es letra: no tocar

# Generator expression: aplica descifrar_chr a cada byte del recorte
# bytes(int, int, int, ...) construye un bytes a partir de ints
recorte = bytes(descifrar_chr(c, numero) for c in recorte)
Cifrado César: 'a' desplazada 3 → 'd'. 'z' desplazada 1 → 'a' (circular). El número X en el stream indica el desplazamiento Y cuántas palabras antes del número hay que descifrar.
Hito 4
Hash MD5 de fichero
El servidor envía un fichero con cabecera de longitud. Hay que calcular su MD5 acumulativo y enviarlo.
Hito 4 MD5 con protocolo longitud:datos TCP — rick:9000
  • 1
    Conecta y envía el identificador
  • 2
    Recibe el primer chunk: b"1234:primerosbytes..."
  • 3
    Separa por b":" con máximo 1 split para obtener longitud y datos
  • 4
    Bucle: hasher.update(chunk), descuenta bytes, pide más hasta llegar a 0
  • 5
    Envía hasher.digest() (16 bytes raw del MD5)
def hito4(ip: str, puerto: int, identificador: bytes) -> bytes:
    hasher = hashlib.md5()

    with socket.socket() as cliente:
        cliente.connect((ip, puerto))
        cliente.sendall(identificador)

        recibido = cliente.recv(1024)

        # Protocolo: "longitud:primerosdatos"
        # split con maxsplit=1 para no partir los datos si tienen ":"
        longitud_b, recibido = recibido.split(b":", 1)
        longitud = int(longitud_b.decode())

        while longitud > 0:
            hasher.update(recibido)            # acumular en el hash
            longitud -= len(recibido)          # descontar bytes procesados
            recibido = cliente.recv(min(longitud, 1024))

        cliente.sendall(hasher.digest())      # 16 bytes raw del MD5
        recibido = cliente.recv(2048)

    return recibido
min(longitud, 1024): evita pedir más bytes de los que faltan. Si faltan 50 bytes, pedimos solo 50. Así el bucle termina limpiamente cuando longitud llega a 0.

Protocolo YAP sobre UDP
Implementación de un protocolo propio con cabecera binaria (struct), checksum y carga codificada en base64.
Hito 5 Protocolo YAP — struct + base64 + checksum UDP — rick:6001
  • 1
    Construye cabecera YAP con checksum=0 usando struct.pack("!3sHBHH", b"YAP", 0, 0, 0, 1)
  • 2
    Codifica el identificador en base64
  • 3
    Calcula cksum(cabecera + carga) y reempaqueta la cabecera con la suma real
  • 4
    Envía por UDP con sendto(), recibe respuesta y desempaqueta
# Estructura del paquete YAP (10 bytes cabecera + carga):
# Bytes 0-2:  "YAP" (numero magico, 3 bytes)
# Bytes 3-4:  tipo  (unsigned short = 2 bytes)
# Byte  5:    codigo (unsigned char = 1 byte)
# Bytes 6-7:  checksum (unsigned short = 2 bytes)
# Bytes 8-9:  secuencia (unsigned short = 2 bytes)
# Bytes 10+:  carga en base64

def preparar_peticion_yap(carga: bytes) -> bytes:
    # Paso 1: cabecera inicial con checksum=0
    cabecera = struct.pack("!3sHBHH", b"YAP", 0, 0, 0, 1)
    # Paso 2: codificar la carga en base64
    carga_codificada = base64.b64encode(carga)
    # Paso 3: calcular checksum del paquete completo
    suma = cksum(cabecera + carga_codificada)
    # Paso 4: reempaquetar con el checksum real
    cabecera = struct.pack("!3sHBHH", b"YAP", 0, 0, suma, 1)
    return cabecera + carga_codificada

def desempaquetar_mensaje_yap(mensaje: bytes) -> bytes:
    # Desempaquetar los primeros 10 bytes = cabecera
    num_mag, tipo, codigo, suma, seq = struct.unpack("!3sHBHH", mensaje[:10])
    carga = mensaje[10:]              # todo lo que hay despues
    return base64.b64decode(carga)    # decodificar base64
El checksum sirve para detectar errores de transmisión. Se calcula sobre el paquete completo (cabecera con checksum=0 + carga). El receptor verifica que el checksum es correcto antes de procesar el paquete.

Consejos para la yinkana
Cambia el usuario: busca usuario: bytes = b"fast_rhino" en main() y pon tu nombre de usuario.
Activa logging DEBUG: con level=logging.DEBUG ves todo lo que se envía y recibe. Imprescindible para depurar.
TCP fragmenta: nunca asumas que un recv() te da el mensaje completo. Usa siempre un bucle con condición de parada.
Bytes everywhere: todo lo que va a la red son bytes. .encode() para convertir str a bytes, .decode() para lo contrario.
Si un hito falla: imprime exactamente lo que recibes con logging.debug(recibido) antes de procesarlo. El protocolo puede haber cambiado entre ediciones.
